1.  WHAT WE ARE USING (Tech Stack)
----------------------------------
- Language: Python 3.9+
- Web Framework: FastAPI (handles API endpoints, auto-documentation at /docs)
- Server: Uvicorn (ASGI server to run FastAPI)
- Database ORM: SQLAlchemy (the "universal translator" for databases)
- Database Engine: SQLite (default). 
    * Why? Zero installation. It's built into Python. Saves to a single file (triage.db).
    * Note: We can switch to PostgreSQL later just by changing one line in .env.
- ML Libraries: LightGBM (for the trained model), NumPy, Pandas.
- Text Mapping: PyYAML + Keyword Matching (fast, zero dependencies, reads config/symptom_mapping.yaml).
    * LLM (Sentence-BERT) is DECOUPLED. We have the adapter code ready, but we skip installing it to avoid dependency hell. The keyword matcher runs in <1ms.
- Environment: python-dotenv (to manage .env variables).
- Scheduler: APScheduler (to run retriage checks every 30 seconds).

# [UPDATED] Production-grade version handling None, Pediatrics, and detailed keywords.
from typing import Optional

def apply_hard_rules(
    age: Optional[float],
    vitals: dict,
    symptom_text: str
):
    """ESI-inspired deterministic safety gate."""
    text = (symptom_text or "").lower().strip()

    def to_float(value):
        try:
            if value is None or value == "":
                return None
            return float(value)
        except (ValueError, TypeError):
            return None

    spo2 = to_float(vitals.get("spo2"))
    hr = to_float(vitals.get("hr"))
    rr = to_float(vitals.get("rr"))
    sbp = to_float(vitals.get("sbp"))
    temp = to_float(vitals.get("temp"))

    # ESI-1 Keywords
    esi_1_keywords = [
        "cardiac arrest", "respiratory arrest", "not breathing",
        "no pulse", "pulseless", "unresponsive", "unconscious"
    ]
    matched_esi_1 = [kw for kw in esi_1_keywords if kw in text]
    if matched_esi_1:
        return {"esi": 1, "source": "hard_gate", "reasons": ["Immediate life-threatening presentation", f"Detected: {', '.join(matched_esi_1)}"]}

    # ESI-1 Extreme Vitals
    if spo2 is not None and spo2 < 85:
        return {"esi": 1, "source": "hard_gate", "reasons": [f"Critically low SpO2 ({spo2}%)"]}
    if sbp is not None and sbp < 70:
        return {"esi": 1, "source": "hard_gate", "reasons": [f"Critically low SBP ({sbp} mmHg)"]}

    # ESI-2 Keywords
    esi_2_keywords = [
        "chest pain", "crushing chest pain", "severe shortness of breath", 
        "respiratory distress", "stroke", "facial droop", "altered mental status", 
        "active seizure", "heavy bleeding", "anaphylaxis", "severe pain"
    ]
    matched_esi_2 = [kw for kw in esi_2_keywords if kw in text]
    if matched_esi_2:
        return {"esi": 2, "source": "hard_gate", "reasons": ["High-risk presentation", f"Detected: {', '.join(matched_esi_2)}"]}

    # ESI-2 Concerning Vitals
    if spo2 is not None and spo2 < 92:
        return {"esi": 2, "source": "hard_gate", "reasons": [f"Low SpO2 ({spo2}%)"]}
    if sbp is not None and sbp < 90:
        return {"esi": 2, "source": "hard_gate", "reasons": [f"Hypotension (SBP {sbp})"]}
    if rr is not None and rr > 30:
        return {"esi": 2, "source": "hard_gate", "reasons": [f"Tachypnea (RR {rr})"]}

    # Pediatric Caution (Fever + Symptoms)
    if age is not None and age < 18:
        pediatric_terms = ["lethargy", "unresponsive", "respiratory distress", "seizure"]
        if any(t in text for t in pediatric_terms):
            return {"esi": 2, "source": "hard_gate", "reasons": ["Pediatric high-risk presentation"]}

    # No hard rules triggered -> proceed to ML
    return {"esi": None, "source": "ml", "reasons": ["Proceed to LightGBM"]}

# [UPDATED] Reads YAML mapping, NO LLM required.
import lightgbm as lgb
import numpy as np
import pandas as pd
import yaml
from pathlib import Path
from .hard_rules import apply_hard_rules

_model = None
_cc_columns = []
_keyword_to_cc = {}

def load_mapping_config():
    global _cc_columns, _keyword_to_cc
    config_path = Path(__file__).parent.parent / "config" / "symptom_mapping.yaml"
    if not config_path.exists():
        raise FileNotFoundError(f"Config not found at {config_path}")
    
    with open(config_path, 'r') as f:
        config = yaml.safe_load(f)
    
    _keyword_to_cc = {}
    cc_set = set()
    for item in config.get('mapping', []):
        _keyword_to_cc[item['phrase'].lower()] = item['cc_column']
        cc_set.add(item['cc_column'])
    _cc_columns = sorted(list(cc_set))
    print(f"[INFO] Loaded {len(_keyword_to_cc)} mappings.")

load_mapping_config()

def load_model():
    global _model
    if _model is None:
        model_path = Path(__file__).parent.parent.parent / "model" / "esi_triage_best_weight7.txt"
        if not model_path.exists():
            raise FileNotFoundError(f"Model not found at {model_path}")
        _model = lgb.Booster(model_file=str(model_path))
    return _model

def text_to_cc_vector(text: str):
    vector = {col: 0 for col in _cc_columns}
    text_lower = text.lower()
    for keyword, col in _keyword_to_cc.items():
        if keyword in text_lower:
            if col in vector:
                vector[col] = 1
    return vector

def predict_esi(vitals: dict, age: int, cc_vector: dict, raw_text: str):
    # Hard Rules
    hard_result = apply_hard_rules(age=float(age), vitals=vitals, symptom_text=raw_text)
    if hard_result["esi"] is not None:
        return hard_result["esi"], 1.0, hard_result["reasons"]

    # Feature Engineering
    features = {}
    features.update(vitals)
    features['age'] = age
    features.update(cc_vector)
    features['n_chief_complaints'] = sum(cc_vector.values())
    features['n_vitals_recorded'] = sum(1 for v in vitals.values() if v is not None and v != "")
    features['has_vitals'] = 1 if features['n_vitals_recorded'] > 0 else 0

    # Predict
    model = load_model()
    df = pd.DataFrame([features])
    raw_score = model.predict(df)[0]
    esi_pred = int(round(raw_score))
    esi_pred = max(1, min(5, esi_pred))

    # Confidence
    boundaries = [1.5, 2.5, 3.5, 4.5]
    min_dist = min(abs(raw_score - b) for b in boundaries)
    confidence = min(1.0, min_dist * 2)

    return esi_pred, confidence, ["ML prediction based on vitals and symptoms"]

from sqlalchemy.orm import Session
from datetime import datetime
from ..models import Queue, Visit

SAFE_WAIT = {1: 0, 2: 600, 3: 1800, 4: 3600, 5: 7200}

def check_retriage(queue_item: Queue, db: Session) -> bool:
    visit = db.query(Visit).filter(Visit.id == queue_item.visit_id).first()
    if not visit:
        return False
    wait_time = (datetime.now() - visit.arrival_time).total_seconds()
    threshold = SAFE_WAIT.get(queue_item.esi_level, 3600)
    return wait_time > threshold

from fastapi import APIRouter, Depends
from sqlalchemy.orm import Session
from datetime import datetime
from ..database import get_db
from ..models import Patient, Visit, Vitals, SymptomCC, Queue
from ..schemas import TriageInput, TriageResponse
from ..ml.model_loader import predict_esi, text_to_cc_vector
from ..services.queue_manager import check_retriage

router = APIRouter(prefix="/triage", tags=["Triage"])

@router.post("/predict", response_model=TriageResponse)
def predict_patient(input: TriageInput, db: Session = Depends(get_db)):
    patient = Patient(name=input.name, age=input.age, has_history=input.has_history)
    db.add(patient)
    db.commit()
    db.refresh(patient)
    
    visit = Visit(patient_id=patient.id)
    db.add(visit)
    db.commit()
    db.refresh(visit)
    
    vitals = Vitals(visit_id=visit.id, **input.vitals.dict())
    db.add(vitals)
    
    cc_vector = text_to_cc_vector(input.symptom_text)
    symptom_cc = SymptomCC(visit_id=visit.id, raw_text=input.symptom_text, features_json=str(cc_vector))
    db.add(symptom_cc)
    
    esi, confidence, reasons = predict_esi(vitals=input.vitals.dict(), age=input.age, cc_vector=cc_vector, raw_text=input.symptom_text)
    
    queue = Queue(visit_id=visit.id, esi_level=esi)
    db.add(queue)
    
    visit.esi_predicted = esi
    visit.esi_final = esi
    visit.confidence_score = confidence
    visit.top_reasons = str(reasons)
    db.commit()
    db.refresh(visit)
    
    return TriageResponse(visit_id=visit.id, esi=esi, confidence=confidence, reasons=reasons, action="Immediate Bed" if esi <= 2 else "Monitor Queue")

@router.get("/queue")
def get_queue(db: Session = Depends(get_db)):
    queue_items = db.query(Queue, Visit).join(Visit).filter(Visit.is_active == True).order_by(
        Queue.esi_level.asc(), Visit.arrival_time.asc()
    ).all()
    
    for q, v in queue_items:
        if check_retriage(q, db):
            q.retriage_needed = True
    db.commit()
    
    result = []
    for q, v in queue_items:
        result.append({
            "queue_id": q.id,
            "visit_id": v.id,
            "patient_id": v.patient_id,
            "esi_level": q.esi_level,
            "wait_time_seconds": (datetime.now() - v.arrival_time).total_seconds(),
            "retriage_needed": q.retriage_needed,
            "confidence": v.confidence_score
        })
    return result

@router.post("/surge/simulate")
def simulate_surge(scale: int = 3, db: Session = Depends(get_db)):
    import random, names
    count = 0
    for _ in range(scale * 30):
        age = random.choice([2, 35, 75])
        vitals = {
            "hr": random.randint(60, 180),
            "sbp": random.randint(80, 160),
            "spo2": random.randint(88, 100),
            "temp": 37.0 + random.random() * 2
        }
        texts = ["chest pain", "abdominal pain", "shortness of breath", "fever", "headache"]
        input = TriageInput(
            name=names.get_first_name(),
            age=age,
            has_history=random.choice([True, False]),
            symptom_text=random.choice(texts),
            vitals=vitals
        )
        predict_patient(input, db)
        count += 1
    return {"message": f"Simulated {count} patients in surge mode"}

# (This remains exactly the same as before)
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.orm import Session
from sqlalchemy.sql import func
from ..database import get_db
from ..models import Visit, Queue, AuditLog
from ..schemas import OverrideInput

router = APIRouter(prefix="/override", tags=["Override"])

@router.put("/visit/{visit_id}")
def override_esi(visit_id: int, input: OverrideInput, db: Session = Depends(get_db)):
    visit = db.query(Visit).filter(Visit.id == visit_id).first()
    if not visit:
        raise HTTPException(404, "Visit not found")
    
    old_esi = visit.esi_final
    visit.esi_final = input.new_esi
    visit.is_overridden = True
    visit.override_reason = input.reason
    visit.overridden_by = input.nurse_id
    visit.override_timestamp = func.now()
    
    queue = db.query(Queue).filter(Queue.visit_id == visit_id).first()
    if queue:
        queue.esi_level = input.new_esi
    
    log = AuditLog(
        visit_id=visit_id,
        action="OVERRIDE",
        old_value=str(old_esi),
        new_value=str(input.new_esi),
        user_id=input.nurse_id,
        reason=input.reason
    )
    db.add(log)
    db.commit()
    
    return {"message": f"ESI updated from {old_esi} to {input.new_esi}"}
